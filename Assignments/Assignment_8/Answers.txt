# Fit a linear model predicting response from predictor
linear_mod <- lm(response ~ predictor, data = df)

# Calculate MSE and plot the straight-line fit over the data
mse_linear <- mean((df$response - fitted(linear_mod))^2)
cat("Linear Model MSE:", round(mse_linear, 4), "\n")

ggplot(df, aes(x = predictor, y = response)) +
  geom_point(alpha = 0.4) +
  geom_smooth(method = "lm", se = TRUE, color = "red") +
  labs(x = "Predictor", y = "Response") +
  theme_minimal()
